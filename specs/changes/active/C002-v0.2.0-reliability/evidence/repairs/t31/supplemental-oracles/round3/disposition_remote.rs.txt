const PAYLOAD: &[u8] = b"remote pinned 9.9.9 payload\n";
fn expected_platform() -> &'static str {
    #[cfg(all(target_os = "macos", target_arch = "aarch64"))]
    {
        "aarch64-apple-darwin"
    }
    #[cfg(all(target_os = "macos", target_arch = "x86_64"))]
    {
        "x86_64-apple-darwin"
    }
    #[cfg(all(target_os = "linux", target_arch = "aarch64"))]
    {
        "aarch64-unknown-linux-gnu"
    }
    #[cfg(all(target_os = "linux", target_arch = "x86_64"))]
    {
        "x86_64-unknown-linux-gnu"
    }
}
const CURL: &str = r#"#!/usr/bin/env python3
import json,sys,os
from pathlib import Path
cfg=json.loads(Path(os.environ['FAKE_CURL_CONFIG']).read_text())
url=sys.argv[-1]
log=Path(os.environ['FAKE_CURL_LOG'])
previous=log.read_text().splitlines() if log.exists() else []
with log.open('a') as f:f.write(url+'\n')
if cfg['mode']=='manifest_failure' and url.endswith('dist-manifest.json') or cfg['mode']=='asset_failure' and not url.endswith('dist-manifest.json'):
 print('controlled curl failure',file=sys.stderr);sys.exit(22)
if cfg['mode']=='stderr_exact':sys.stderr.buffer.write(b'x'*(1024*1024));sys.stderr.buffer.flush()
platform=cfg['platform']
version=cfg.get('version','9.9.9')
if url.endswith('/latest/download/dist-manifest.json'):
 if any('/latest/download/' in old for old in previous):version='8.8.8'
elif not url.startswith(cfg['base']+'/download/v'+version+'/'):
 print('unexpected URL '+url,file=sys.stderr);sys.exit(22)
name='sheltie-'+version+'-'+platform
if url.endswith('dist-manifest.json'):
 digest='dec2fc95c56b937d66a20cf64c623531d22fdd4def9a52dfadd4ebec7a60a694'
 print(json.dumps({'version':version,'assets':[{'platform':platform,'name':name,'sha256':digest}]}))
elif url==cfg['base']+'/download/v'+version+'/'+name:
 sys.stdout.buffer.write(b'remote pinned 9.9.9 payload\n')
else:
 print('unexpected asset '+url,file=sys.stderr);sys.exit(22)
"#;
struct Transport {
    _dir: tempfile::TempDir,
    tool: PathBuf,
    config: PathBuf,
    log: PathBuf,
}

// Task: C002-T31
#[test]
fn supplemental_remote_invalid_version_is_rejected_before_transport() {
    for version in ["", ".", "..", "bad/tag", "bad\\tag"] {
        let env = Env::new();
        env.ok(&["self", "install"]);
        let home = env.dir.path();
        let old = read(home, "bin/sheltie");
        let store = read(home, "store.db");
        let transport = Transport::new("normal");
        let mut config: Value = serde_json::from_slice(&std::fs::read(&transport.config).unwrap()).unwrap();
        config["version"] = json!(version);
        std::fs::write(&transport.config, config.to_string()).unwrap();
        let out = transport.update(&env, Some(version));
        let response: Value = serde_json::from_slice(&out.stdout).unwrap();
        assert_eq!(response["error"]["code"], "UPDATE_UNAVAILABLE");
        assert!(!transport.log.exists(), "版本拒绝必须发生在任何传输之前");
        assert_eq!(read(home, "bin/sheltie"), old);
        assert_eq!(read(home, "store.db"), store);
        assert!(!home.join("bin/sheltie.prev").exists());
    }
}

// Task: C002-T31
#[test]
fn supplemental_install_creates_an_absent_home_and_a_wal_store() {
    let parent = tempfile::tempdir().unwrap();
    let home = parent.path().join("new-home");
    assert!(!home.exists());
    let out = Command::new(env!("CARGO_BIN_EXE_sheltie"))
        .arg("--home").arg(&home).args(["--json", "self", "install"]).output().unwrap();
    assert!(out.status.success(), "{}", String::from_utf8_lossy(&out.stdout));
    assert!(home.join("bin/sheltie").is_file());
    let conn = rusqlite::Connection::open_with_flags(home.join("store.db"), rusqlite::OpenFlags::SQLITE_OPEN_READ_ONLY).unwrap();
    let version: i64 = conn.query_row("PRAGMA user_version", [], |r| r.get(0)).unwrap();
    let journal: String = conn.query_row("PRAGMA journal_mode", [], |r| r.get(0)).unwrap();
    assert_eq!(version, 2);
    assert_eq!(journal, "wal");
}

// Task: C002-T31
#[test]
fn supplemental_purge_rejects_all_hardlinks_before_removing_any_root() {
    let env = Env::new();
    env.ok(&["self", "install"]);
    let outside = tempfile::tempdir().unwrap();
    let sentinel = outside.path().join("sentinel");
    std::fs::write(&sentinel, b"outside independent bytes").unwrap();
    let home = env.dir.path();
    std::fs::write(home.join("a-owned"), b"owned independent bytes").unwrap();
    std::fs::hard_link(&sentinel, home.join("z-hardlink")).unwrap();
    let binary = read(home, "bin/sheltie");
    let store = read(home, "store.db");
    let out = env.cmd(&["self", "uninstall", "--purge", "--yes"]).output().unwrap();
    assert!(!out.status.success());
    assert_eq!(read(home, "a-owned"), b"owned independent bytes");
    assert_eq!(read(home, "z-hardlink"), b"outside independent bytes");
    assert_eq!(std::fs::read(&sentinel).unwrap(), b"outside independent bytes");
    assert_eq!(read(home, "bin/sheltie"), binary);
    assert_eq!(read(home, "store.db"), store);
}

// Task: C002-T31
#[test]
fn supplemental_rollback_rejects_a_hardlinked_previous_binary() {
    let env = Env::new();
    env.ok(&["self", "install"]);
    let home = env.dir.path();
    std::fs::remove_file(home.join("bin/sheltie")).unwrap();
    let outside = tempfile::tempdir().unwrap();
    let sentinel = outside.path().join("sentinel");
    std::fs::write(&sentinel, b"hardlinked old binary").unwrap();
    std::fs::hard_link(&sentinel, home.join("bin/sheltie.prev")).unwrap();
    let store = read(home, "store.db");
    let out = env.cmd(&["self", "rollback"]).output().unwrap();
    assert!(!out.status.success());
    assert!(!home.join("bin/sheltie").exists());
    assert_eq!(read(home, "bin/sheltie.prev"), b"hardlinked old binary");
    assert_eq!(std::fs::read(&sentinel).unwrap(), b"hardlinked old binary");
    assert_eq!(read(home, "store.db"), store);
}
impl Transport {
    fn new(mode: &str) -> Self {
        let dir = tempfile::tempdir().unwrap();
        let tool = dir.path().join("tools");
        std::fs::create_dir(&tool).unwrap();
        let executable = tool.join("curl");
        std::fs::write(&executable, CURL).unwrap();
        std::fs::set_permissions(executable, std::fs::Permissions::from_mode(0o755)).unwrap();
        let config = dir.path().join("config.json");
        std::fs::write(
            &config,
            json!({"mode":mode,"base":BASE,"platform":expected_platform()}).to_string(),
        )
        .unwrap();
        let log = dir.path().join("requests.txt");
        Self {
            _dir: dir,
            tool,
            config,
            log,
        }
    }
    fn update(&self, env: &Env, version: Option<&str>) -> Output {
        let mut cmd = Command::new(env!("CARGO_BIN_EXE_sheltie"));
        cmd.args(["--home", &env.home(), "--json", "self", "update"]);
        if let Some(version) = version {
            cmd.args(["--version", version]);
        }
        cmd.env("SHELTIE_RELEASE_BASE", BASE)
            .env("FAKE_CURL_CONFIG", &self.config)
            .env("FAKE_CURL_LOG", &self.log)
            .env(
                "PATH",
                format!("{}:{}", self.tool.display(), std::env::var("PATH").unwrap()),
            );
        cmd.output().unwrap()
    }
    fn requests(&self) -> Vec<String> {
        std::fs::read_to_string(&self.log)
            .unwrap()
            .lines()
            .map(str::to_owned)
            .collect()
    }
}
fn read(home: &Path, name: &str) -> Vec<u8> {
    std::fs::read(home.join(name)).unwrap()
}

// Task: C002-T31
#[test]
fn remote_update_pins_latest_and_explicit_tags_and_accepts_exact_stderr_limit() {
    for (mode, version) in [
        ("normal", None),
        ("normal", Some("9.9.9")),
        ("stderr_exact", None),
    ] {
        let env = Env::new();
        env.ok(&["self", "install"]);
        let home = env.dir.path();
        let old = read(home, "bin/sheltie");
        let store = read(home, "store.db");
        let transport = Transport::new(mode);
        let out = transport.update(&env, version);
        assert!(
            out.status.success(),
            "{}",
            String::from_utf8_lossy(&out.stdout)
        );
        let response: Value = serde_json::from_slice(&out.stdout).unwrap();
        assert_eq!(response["data"]["to"], "9.9.9");
        let mut expected = Vec::new();
        if version.is_none() {
            expected.push(format!("{BASE}/latest/download/dist-manifest.json"));
        }
        expected.push(format!("{BASE}/download/v9.9.9/dist-manifest.json"));
        expected.push(format!(
            "{BASE}/download/v9.9.9/sheltie-9.9.9-{}",
            expected_platform()
        ));
        assert_eq!(transport.requests(), expected);
        assert_eq!(read(home, "bin/sheltie"), PAYLOAD);
        assert_eq!(read(home, "bin/sheltie.prev"), old);
        assert_eq!(read(home, "store.db"), store);
    }
}

// Task: C002-T31
#[test]
fn failed_remote_manifest_or_asset_preserves_binaries_store_and_cleans_its_tmp() {
    for mode in ["manifest_failure", "asset_failure"] {
        let env = Env::new();
        env.ok(&["self", "install"]);
        let home = env.dir.path();
        std::fs::write(home.join("bin/sheltie.prev"), b"previous independent bytes").unwrap();
        let old = read(home, "bin/sheltie");
        let prev = read(home, "bin/sheltie.prev");
        let store = read(home, "store.db");
        let transport = Transport::new(mode);
        let out = transport.update(&env, None);
        assert!(!out.status.success());
        let response: Value = serde_json::from_slice(&out.stdout).unwrap();
        assert_eq!(response["error"]["code"], "UPDATE_UNAVAILABLE");
        assert_eq!(read(home, "bin/sheltie"), old);
        assert_eq!(read(home, "bin/sheltie.prev"), prev);
        assert_eq!(read(home, "store.db"), store);
        assert_eq!(std::fs::read_dir(home.join("tmp")).unwrap().count(), 0);
    }
}
