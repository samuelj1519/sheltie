#![allow(clippy::unwrap_used, clippy::expect_used)]
mod common;
use common::*;
use rusqlite::{Connection, types::Value as SqlValue};
use serde_json::{Value, json};

fn effects(conn: &Connection, rid: &str) -> Value {
    let raw: String = conn
        .query_row(
            "SELECT effects_json FROM requests WHERE request_id=?1",
            [rid],
            |r| r.get(0),
        )
        .unwrap();
    serde_json::from_str(&raw).unwrap()
}
fn set_effects(conn: &Connection, rid: &str, effects: &Value, published: i64) {
    conn.execute(
        "UPDATE requests SET effects_json=?1,published=?2 WHERE request_id=?3",
        rusqlite::params![effects.to_string(), published, rid],
    )
    .unwrap();
}
type StoreRows = Vec<Vec<Vec<SqlValue>>>;
type BusinessFiles = BTreeMap<PathBuf, (u32, Vec<u8>)>;

fn rows(conn: &Connection) -> StoreRows {
    ["workbooks", "works", "work_sequence", "requests", "audit"]
        .into_iter()
        .map(|table| {
            let mut q = conn
                .prepare(&format!("SELECT * FROM {table} ORDER BY rowid"))
                .unwrap();
            let n = q.column_count();
            q.query_map([], |r| (0..n).map(|i| r.get(i)).collect())
                .unwrap()
                .map(Result::unwrap)
                .collect()
        })
        .collect()
}
fn files(root: &Path) -> BusinessFiles {
    fn walk(base: &Path, dir: &Path, out: &mut BusinessFiles) {
        if !dir.exists() {
            return;
        }
        let meta = std::fs::symlink_metadata(dir).unwrap();
        let bytes = if meta.is_file() {
            std::fs::read(dir).unwrap()
        } else {
            Vec::new()
        };
        out.insert(
            dir.strip_prefix(base).unwrap().to_path_buf(),
            (meta.permissions().mode(), bytes),
        );
        if meta.is_dir() {
            for e in std::fs::read_dir(dir).unwrap() {
                walk(base, &e.unwrap().path(), out);
            }
        }
    }
    let mut out = BTreeMap::new();
    for name in ["works", "workbooks", "pending"] {
        walk(root, &root.join(name), &mut out);
    }
    out
}
fn assert_unchanged(
    home: &sheltie_runtime::Home,
    conn: &Connection,
    before_rows: StoreRows,
    before_files: BusinessFiles,
) {
    assert_eq!(rows(conn), before_rows, "拒绝前不得提交或改变完成标记");
    assert_eq!(
        files(home.root().as_path().as_std_path()),
        before_files,
        "整批登记校验失败不得写文件、封存或清理元数据"
    );
}

fn assert_original_data(snapshot: &Value, expected: &Value) {
    let mut data = snapshot["data"].clone();
    assert_eq!(
        data.as_object_mut().unwrap().remove("replayed"),
        Some(json!(false))
    );
    assert_eq!(&data, expected);
}

#[test]
fn supplemental_managed_rename_rejects_a_hardlinked_regular_source() {
    use sheltie_runtime::fsx::{ManagedFs, ManagedRelPath};
    let (dir, home, _svc) = home_with_example("two-step");
    let fs = ManagedFs::open_existing(&home).unwrap();
    let lock = home.acquire_lock().unwrap();
    let original = ManagedRelPath::new("tmp/regular-source").unwrap();
    let moved = ManagedRelPath::new("tmp/regular-moved").unwrap();
    fs.write_new(&lock, &original, b"single linked positive").unwrap();
    fs.rename_new(&lock, &original, &moved).unwrap();
    assert_eq!(std::fs::read(dir.path().join(moved.as_str())).unwrap(), b"single linked positive");
    let source = ManagedRelPath::new("tmp/hardlink-source").unwrap();
    let target = ManagedRelPath::new("tmp/hardlink-target").unwrap();
    fs.write_new(&lock, &source, b"independent linked bytes").unwrap();
    let outside = tempfile::tempdir().unwrap();
    let alias = outside.path().join("alias");
    std::fs::hard_link(dir.path().join(source.as_str()), &alias).unwrap();
    let error = fs.rename_new(&lock, &source, &target).unwrap_err();
    assert!(matches!(error, sheltie_runtime::Error::InvalidRequest { .. }), "{error:?}");
    assert!(!dir.path().join(target.as_str()).exists());
    assert_eq!(std::fs::read(dir.path().join(source.as_str())).unwrap(), b"independent linked bytes");
    assert_eq!(std::fs::read(&alias).unwrap(), b"independent linked bytes");
}

#[test]
fn completed_workbook_replay_checks_effect_shape_identity_and_pending_path() {
    for remove in [false, true] {
        let (_dir, home) = temp_home();
        let repo = repo(&home);
        let source = abs(&example_dir("two-step"));
        let rid = if remove { "remove-shape" } else { "add-shape" };
        let added = repo
            .add(&source, Some(if remove { "setup-add" } else { rid }.into()))
            .unwrap();
        let expected_data = if remove {
            repo.remove("two-step", "1.0.0", Some(rid.into()))
                .unwrap()
                .data
        } else {
            added.data
        };
        if remove {
            assert!(
                repo.remove("two-step", "1.0.0", Some(rid.into()))
                    .unwrap()
                    .replayed
            );
        } else {
            assert!(repo.add(&source, Some(rid.into())).unwrap().replayed);
        }
        let conn = Connection::open(home.store_path().as_str()).unwrap();
        let pristine = effects(&conn, rid);
        let pristine_reply: String = conn
            .query_row(
                "SELECT reply_json FROM requests WHERE request_id=?1",
                [rid],
                |r| r.get(0),
            )
            .unwrap();
        for corruption in [
            "empty_batch",
            "final",
            "owner",
            "digest",
            "digest_format",
            "digest_root",
            "pending_prefix",
            "pending_leaf",
            "pending_uuid",
            "shared_bad_digest",
        ] {
            if remove && matches!(corruption, "digest_root" | "digest" | "shared_bad_digest") {
                continue;
            }
            let mut changed = pristine.clone();
            if corruption == "empty_batch" {
                changed = json!([]);
            } else {
                let op = &mut changed[0];
                assert!(op.get("final").is_some() && op.get("final_path").is_none());
                match corruption {
                    "final" => op["final"] = json!("workbooks/other/1.0.0"),
                    "owner" => op["owner"] = json!("workbook:other@1.0.0"),
                    "digest" => {
                        assert_ne!(op["digest"], json!("0".repeat(64)));
                        op["digest"] = json!("0".repeat(64));
                    }
                    "digest_format" => op["digest"] = json!("invalid-digest"),
                    "shared_bad_digest" => {
                        // 两份相等的摘要仍必须满足SHA-256格式；此事实已损坏，不能返回可信original。
                        op["digest"] = json!("invalid-digest");
                        let mut reply: Value = serde_json::from_str(&pristine_reply).unwrap();
                        reply["data"]["digest"] = json!("invalid-digest");
                        conn.execute(
                            "UPDATE requests SET reply_json=?1 WHERE request_id=?2",
                            rusqlite::params![reply.to_string(), rid],
                        )
                        .unwrap();
                    }
                    "digest_root" => op["digest_root"] = json!("workbook"),
                    "pending_prefix" => {
                        op["pending"] = json!(
                            op["pending"]
                                .as_str()
                                .unwrap()
                                .replacen("pending/", "other/", 1)
                        )
                    }
                    "pending_leaf" => {
                        op["pending"] = json!(
                            op["pending"]
                                .as_str()
                                .unwrap()
                                .replace("/payload", "/other")
                        )
                    }
                    "pending_uuid" => op["pending"] = json!("pending/invalid/payload"),
                    _ => unreachable!(),
                }
            }
            set_effects(&conn, rid, &changed, 1);
            let before_rows = rows(&conn);
            let before_files = files(home.root().as_path().as_std_path());
            let error = if remove {
                repo.remove("two-step", "1.0.0", Some(rid.into()))
                    .unwrap_err()
            } else {
                repo.add(&source, Some(rid.into())).unwrap_err()
            };
            if corruption == "shared_bad_digest" {
                assert_effect_pending_without_original(error, true, rid, None);
            } else {
                let snapshot = assert_effect_pending(error, true, Some(rid), None);
                assert_eq!(snapshot["request_id"], rid);
                assert_original_data(&snapshot, &expected_data);
            }
            assert_unchanged(&home, &conn, before_rows, before_files);
            set_effects(&conn, rid, &pristine, 1);
            conn.execute(
                "UPDATE requests SET reply_json=?1 WHERE request_id=?2",
                rusqlite::params![pristine_reply, rid],
            )
            .unwrap();
        }
    }
}

fn cleanup_checks_the_entire_request_index_before_removing_a_legitimate_orphan() {
    let (_dir, home) = temp_home();
    let repo = repo(&home);
    let rid = "index-publisher";
    repo.add(&abs(&example_dir("two-step")), Some(rid.into()))
        .unwrap();
    let svc = service(&home);

        .begin(
            &work,

            Some("index-begin".into()),
        )
        .unwrap();
    let original_brief = std::fs::read(begun.data["brief_path"].as_str().unwrap()).unwrap();
    std::fs::write(
        output_dir_of(&begun).as_path().join("outline.md"),
        b"index output",
    )
    .unwrap();
    svc.submit(
        &work,
        &AttemptId::parse("outline#1.0").unwrap(),
        &InputValue::Literal {
            text: "done".into(),
        },
        Some("index-submit".into()),
    )
    .unwrap();
    let conn = Connection::open(home.store_path().as_str()).unwrap();
    let pristine_publish = effects(&conn, rid);
    let pristine_begin = effects(&conn, "index-begin");
    let pristine_start = effects(&conn, &started.request_id);
    let pristine_submit = effects(&conn, "index-submit");
    let id = "0198f01a7f00700080000000000000fa";
    let pending = home.pending_dir();
    let private = pending.as_path().join(id).join("payload/private");
    std::fs::create_dir_all(private.parent().unwrap()).unwrap();
    std::fs::write(&private, b"legitimate orphan original").unwrap();
    std::fs::write(pending.as_path().join(format!("{id}.owner")),json!({"format":"pending/v1","internal_id":id,"request_id":"uncommitted-index-neighbor","op":"add_workbook"}).to_string()+"\n").unwrap();
    for corruption in [
        "publish_final",
        "publish_owner",
        "publish_digest",
        "publish_root",
        "publish_version",
        "workbook_work_id",
        "history_ancestor",
        "work_publish_root",
        "seal_digest",
    ] {
        let (changed_request, pristine) = match corruption {
            "history_ancestor" => ("index-begin", &pristine_begin),
            "work_publish_root" => (started.request_id.as_str(), &pristine_start),
            "seal_digest" => ("index-submit", &pristine_submit),
            _ => (rid, &pristine_publish),
        };
        let mut changed = pristine.clone();
        match corruption {
            "publish_final" => changed[0]["final"] = json!("workbooks/other/1.0.0"),
            "publish_owner" => changed[0]["owner"] = json!("workbook:other@1.0.0"),
            "publish_digest" => changed[0]["digest"] = json!("invalid-digest"),
            "publish_root" => changed[0]["digest_root"] = json!("workbook"),
            "publish_version" => {
                // 一个Workbook版本事实在owner与final中的两份表示保持一致，只破坏其格式。
                changed[0]["owner"] = json!("workbook:two-step@bad_");
                changed[0]["final"] = json!("workbooks/two-step/bad_");
            }
            "work_publish_root" => changed[0]["digest_root"] = json!(""),
            "seal_digest" => {
                changed
                    .as_array_mut()
                    .unwrap()
                    .iter_mut()
                    .find(|op| op["kind"] == "seal_outputs")
                    .unwrap()["refs"][0]["sha256"] = json!("invalid-digest")
            }
            "workbook_work_id" => {
                conn.execute(
                    "UPDATE requests SET work_id=?1 WHERE request_id=?2",
                    rusqlite::params![work.as_str(), rid],
                )
                .unwrap();
            }
            "history_ancestor" => {
                changed
                    .as_array_mut()
                    .unwrap()
                    .iter_mut()
                    .find(|op| op["kind"] == "write_file")
                    .unwrap()["path"] = json!("works")
            }
            _ => unreachable!(),
        }
        set_effects(&conn, changed_request, &changed, 1);
        let before_rows = rows(&conn);
        let before_files = files(home.root().as_path().as_std_path());
        let error = repo.cleanup_pending().unwrap_err();
        assert_eq!(
            error.code(),
            sheltie_core::ErrorCode::StoreCorrupt,
            "{corruption}: {error:?}"
        );
        assert_unchanged(&home, &conn, before_rows, before_files);
        set_effects(&conn, rid, &pristine_publish, 1);
        set_effects(&conn, "index-begin", &pristine_begin, 1);
        set_effects(&conn, &started.request_id, &pristine_start, 1);
        set_effects(&conn, "index-submit", &pristine_submit, 1);
        conn.execute(
            "UPDATE requests SET work_id=NULL WHERE request_id=?1",
            [rid],
        )
        .unwrap();
    }
    assert!(repo.cleanup_pending().unwrap().is_empty());
    assert!(!private.exists());
    assert_eq!(
        std::fs::read(begun.data["brief_path"].as_str().unwrap()).unwrap(),
        original_brief
    );
}
