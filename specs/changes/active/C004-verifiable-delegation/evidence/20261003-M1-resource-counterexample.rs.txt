use sheltie_runtime::{Home, WorkService, WorkbookRepo, StartArgs};
use sheltie_runtime::request::InputValue;
use serde_json::{Value, json};
use rusqlite::Connection;
use std::collections::BTreeMap;
fn main() {
    let base = std::path::PathBuf::from(format!("/private/tmp/sheltie-c004-review-resource-{}",std::process::id()));
    for (kind, source) in [("resource","resource.resources/reference.txt"),("stats","engine.stats")] {
        let root = base.join(kind);
        let workbook = root.join("source");
        std::fs::create_dir_all(workbook.join("flows")).unwrap();
        std::fs::create_dir_all(workbook.join("resources")).unwrap();
        std::fs::write(workbook.join("resources/reference.txt"), b"frozen resource bytes\n").unwrap();
        std::fs::write(workbook.join("workbook.toml"), "schema='workbook/v1'\nid='results'\nversion='1.0.0'\nname='Results'\nflows=['flows/default.toml']\n").unwrap();
        std::fs::write(workbook.join("flows/default.toml"), format!("schema='flow/v1'\nid='default'\nentry='deliver'\n[[nodes]]\nid='deliver'\ntitle='Deliver'\nexecutor='agent'\ninstruction={{text='Deliver reference.'}}\ninputs=[{{name='reference',from='{source}',result=true}}]\noutputs=[]\n")).unwrap();
        let home = Home::resolve(Some(root.join("home").to_str().unwrap())).unwrap();
        let repo = WorkbookRepo::new(home.clone());
        repo.add(&serde_json::from_value(json!(workbook.to_str().unwrap())).unwrap(), None).unwrap();
        let service = WorkService::new(home.clone());
        let response = service.start(StartArgs {workbook_id:"results".into(),version:None,flow:"default".into(),name:None,inputs:BTreeMap::new()},None).unwrap();
        let work = serde_json::from_value(response.data["work_id"].clone()).unwrap();
        let node = serde_json::from_value(json!("deliver")).unwrap();
        let begun = service.begin(&work,&node,None).unwrap();
        let attempt = serde_json::from_value(serde_json::to_value(&begun.reply).unwrap()["attempt"].clone()).unwrap();
        service.submit(&work,&attempt,&InputValue::Literal{text:"complete".into()},None).unwrap();
        let before = service.result(&work).unwrap().0;
        println!("{kind} before final={} digest={} bytes={}", before.r#final,before.artifacts[0].sha256.as_str(),before.artifacts[0].bytes);
        let conn = Connection::open(home.store_path().as_str()).unwrap();
        let raw:String = conn.query_row("SELECT state_json FROM works",[],|r|r.get(0)).unwrap();
        for field in ["sha256","bytes"] {
            let mut state:Value = serde_json::from_str(&raw).unwrap();
            state["attempts"][0]["inputs"]["reference"][field] = if field == "bytes" { json!(999) } else { json!("0".repeat(64)) };
            conn.execute("UPDATE works SET state_json=?1",[state.to_string()]).unwrap();
            match service.result(&work) {
                Ok((view,_)) => println!("{kind} modified {field}: SUCCESS final={} digest={} bytes={}",view.r#final,view.artifacts[0].sha256.as_str(),view.artifacts[0].bytes),
                Err(error) => println!("{kind} modified {field}: ERROR {:?} {error}", error.code()),
            }
        }
    }
}
